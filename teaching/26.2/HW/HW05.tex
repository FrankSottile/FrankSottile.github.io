\item  \label{Exer:Two_subgroups}
  Let $G$ be a group and $a\in G$.
  Define the following subsets of $G$,
  \[
  \defcolor{C_G(a)}\ \vcentcolon=\ \{ g\in G\mid ag=ga\}
  \qquad\mbox{and}\qquad
  \defcolor{Z(G)}\ \vcentcolon=\ \{ z\in G\mid zg=gz\ \forall g\in G\}\,.
  \]
  Prove that these are subgroups of $G$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\item
  For each of the subgroups $H$ of $G$ defined in Exercise~\ref{Exer:Two_subgroups}, prove or give a counterexample
  that $H$ is abelian.

  What is the relation between $C_G(a)$ and $Z(G)$?
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\item
  Let $G$ be a finite cyclic group with subgroups $H$ and $K$.
  Prove that $H=K$ if and only if $|H|=|K|$.

  Does this hold if $G$ is an infinite cyclic group?
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\item
  Give an example of a finite group $G$ and subgroups $H,K$ of $G$ with $|H|=|K|$ but $H\not\simeq K$.
  (E.g.~$H$ and $K$ are not isomorphic.)
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\item
  Let $a$ and $b$ be elements of a group $G$.
  Prove that the two elements $ab$ and $ba$ of $G$ have the same order.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\item
  Consider $D_4$, the dihedral group of order 8.
  Find two elements $\mu$, $\nu$ of $D_4$ of order two which generate $D_4$.
  Draw a Cayley graph of $D_4$ with these generators.

  Generalize this to $D_n$ (e.g.\ for $n>2$, describe the Cayley graph).
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\item
  Let $A_4$ be the alternating group of even permutations in the symmetric group $S_4$.
  \begin{enumerate}
   \item List the elements of $A_4$.
   \item Let $S\vcentcolon=\{(1,2,3)\,,\, (1,2)(3,4) \}$, which generates $A_4$.
        Draw the Cayley digraph of $(A_4, S)$, as nice as possible.
   \item List all of the subgroups of $A_4$.
  \end{enumerate}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\item
  Let $\varphi\colon G\to H$ be a morphism of groups.                    %morphism is a synonym for homomorphism
  Prove that $\varphi$ is injective if and only if $\ker(\varphi)=\{e\}$.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\end{enumerate}
