%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
 \item
   Let $S$ be a set and $B\vcentcolon = 2^S$, the collection of subsets of $S$.
   For subsets $X$, $Y$ of $S$, define
   \[
   X+Y\ \vcentcolon =\ X\cup Y \smallsetminus X\cap Y
   \qquad\mbox{and}\qquad
   X \cdot Y \vcentcolon=\ X\cap Y\,.
   \]
   Show that with these operations $B$ forms a ring.
 
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
 \item  We discussed how the unit quaternions act on $\RR^3=\RR\bi + \RR\bj+\RR\bk$ by conjugation.
   Give $3\times 3$ matrices (in the ordered basis $\bi,\bj,\bk$ for the action) of the unit quaternions:
   \[
   1\,,\ \bi\,,\ \bj\,,\ \bk\,,\
   \sin(\phi)\cos(\theta)\bi+ \sin(\phi)\sin(\theta)\bj+ \cos(\phi)\bk\,,\
   a+b\bi+c\bj+d\bk\,.
   \]
   (For the last, note that $a^2+b^2+c^2+d^2=1$, as it is a unit quaternion.)
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%



%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
 \item  Let $R$ be a ring with the following property:
   For all $r\in R$, there is an $s\in R$ with $rsr=r$.
   \begin{enumerate}
     \item Show that $R$ has no zero-divisors.
     \item Show that $srs=s$ (where $r,s$ are as above).
     \item Show that $R$ is a division ring.
   \end{enumerate}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%



%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
 \item  An element of a ring $R$ is \demph{idempotent} if $e^e=e$
   \begin{enumerate}
   \item  Show that if a ring $R$ consists only of idempotent elements, then $R$ is commuative.
   \item  Show that a division ring $D$ has exactly two idempotent elements.
   \item  Let $R=\Mat_{3\times 3}(\RR)$.   Exhibit a non-zero, non-identity idempotent $e$ in $R$.
   \item  Let $e$ be an idempotent in a ring $R$.
     Show that $1-e$ is an idempotent and that $e(1-e)=0$.
   \end{enumerate}
 
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
 \item  Let $p$ be a prime.
   Show that $1$ and $p-1$ are the only elements of $\ZZ_p=\FF_p$ that are their own multiplicative inverses.

 
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%




\end{enumerate}
